cmake_minimum_required(VERSION 3.22)
project(avid_core_jni C)
if(NOT ANDROID)
  message(FATAL_ERROR "Use the Android NDK toolchain")
endif()
if(ANDROID_PLATFORM_LEVEL LESS 26)
  message(FATAL_ERROR "AVID linked profile requires Android API 26+")
endif()
if(NOT DEFINED AVID_ARTIFACT_DIR)
  message(FATAL_ERROR "Set AVID_ARTIFACT_DIR to a verified linked artifact for ANDROID_ABI")
endif()
add_library(avid_core SHARED IMPORTED)
set_target_properties(avid_core PROPERTIES IMPORTED_LOCATION "${AVID_ARTIFACT_DIR}/lib/libavid_core_ffi.so")
add_library(avid_core_jni SHARED linked_jni.c)
target_include_directories(avid_core_jni PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../include")
target_link_libraries(avid_core_jni PRIVATE avid_core)
target_link_options(avid_core_jni PRIVATE "-Wl,-z,max-page-size=16384")
